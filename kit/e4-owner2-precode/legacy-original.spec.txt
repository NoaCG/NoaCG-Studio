// Ignored one-shot diagnostic; frozen63d7 raw legacy target boundary.
import { test, expect } from '@playwright/test';
import { writeFileSync } from 'node:fs';
import { seedAuthoringProject } from '../_authoringProject';
import { durableValue } from '../_storage';

test('raw legacy production installation uses the supported read identity', async ({ page }) => {
  await seedAuthoringProject(page, ['lt01']);
  const seeded = await page.evaluate(async () => {
    const A = await import('/src/store/saveActions.ts');
    const D = await import('/src/model/durableStore.ts');
    const S = await import('/src/model/shows.ts');
    const saved = await A.saveGraphicAs('Legacy diagnostic source', { kind: 'standalone' });
    if (!saved.ok) throw new Error(saved.error!);
    (await import('/src/store/templateStore.ts')).flushPendingProject();
    await D.commitDurableWrites();
    const raw = [{ id: 'owner2-legacy', name: 'Untouched legacy target', graphics: [] },
      { id: 'owner2-unrelated', name: 'Untouched other legacy production', graphics: [], marker: 'preserve raw representation' }];
    D.durable.setItem('spx-gfx-shows', JSON.stringify(raw));
    await D.commitDurableWrites();
    const normalized = S.loadShows().find(s => s.id === raw[0].id)!;
    return { raw, normalized, graphicId: saved.graphicId! };
  });
  const before = JSON.parse((await durableValue(page, 'spx-gfx-shows'))!);
  expect(before).toEqual(seeded.raw);
  expect(seeded.normalized.version).toBe(2);
  expect(seeded.normalized.updatedAt).toBe('1970-01-01T00:00:00.000Z');
  const result = await page.evaluate(async graphicId => {
    try {
      const plan = await (await import('/src/control/projectInstallation.ts')).installProjectGraphics('owner2-legacy', [graphicId]);
      return { status: plan.status, error: '' };
    } catch (e) { return { status: 'failed', error: String(e) }; }
  }, seeded.graphicId);
  const after = JSON.parse((await durableValue(page, 'spx-gfx-shows'))!);
  writeFileSync(test.info().outputPath('raw-legacy.json'), JSON.stringify({ source: '63d7d7b57253f49cdcfcaa20a5805dbdcc8c2baf', seeded, before, result, after }, null, 2));
  expect(after.find((s: { id: string }) => s.id === 'owner2-unrelated')).toEqual(seeded.raw[1]);
  expect(result, 'an unchanged supported legacy target must install without a concurrent-edit conflict').toEqual({ status: 'confirmed', error: '' });
});
