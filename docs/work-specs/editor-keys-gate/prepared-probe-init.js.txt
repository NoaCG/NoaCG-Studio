() => {
  if (window !== window.top) return;
  const clone = value => JSON.parse(JSON.stringify(value));
  const near = (a, b) => typeof a === 'number' && Math.abs(a - b) < .00001;
  const sameRevision = (a, b) => a?.source === b?.source && a?.assets === b?.assets;
  const state = { installed: true, connected: false, armed: false, ruler: false,
    events: [], problems: [], snapshot: null, zero: null, one: null,
    zeroHeldBeforeController: false, zeroHeldAfterMicrotask: false,
    zeroReplayConsumed: false, oneHeldBeforeController: false,
    oneHeldAfterMicrotask: false, latestReleased: false };
  window.__keysProbe = state;
  const replay = new WeakSet();
  const timers = new Set();
  const later = (fn, ms) => { const id = setTimeout(() => { timers.delete(id); fn(); }, ms); timers.add(id); };
  const snapshot = () => state.snapshot ? clone(state.snapshot()) : null;
  const dom = () => {
    const canvas = document.querySelector('[data-testid="foundation-canvas"]');
    return Object.fromEntries(['data-pending', 'data-request', 'data-generation', 'data-pose-time', 'data-pose-cue']
      .map(name => [name, canvas?.getAttribute(name) ?? null]));
  };
  const summary = event => ({ documentId: event.data.documentId, generation: event.data.generation,
    requestId: event.data.requestId, revision: event.data.revision, kind: event.data.kind, renderedAt: event.data.renderedAt ?? null,
    parts: event.data.parts?.map(part => ({ selector: part.selector, time: part.appearance?.time ?? null,
      cue: part.appearance?.cue ?? null, revision: part.appearance?.revision ?? null })) ?? [] });
  const record = (kind, event) => state.events.push(clone({ kind,
    at: performance.timeOrigin + performance.now(), dom: dom(), session: snapshot(),
    reply: event ? summary(event) : null, origin: event?.origin ?? null,
    sourceMatchesFrame: event ? event.source === document.querySelector('iframe[title="Foundation graphic preview"]')?.contentWindow : null }));
  const expectedActual = (event, observed) => {
    const frame = document.querySelector('iframe[title="Foundation graphic preview"]');
    const message = event.data, expected = observed?.expected;
    return event.source === frame?.contentWindow && expected &&
      message.documentId === expected.documentId && message.generation === expected.generation &&
      message.requestId === expected.requestId && sameRevision(message.revision, expected.revision) &&
      observed.view.selectedParts.length === 1 && observed.view.selectedParts[0] === '#nested' &&
      sameRevision(message.revision, observed.version);
  };
  const unconsumed = (event, observed) => expectedActual(event, observed) &&
    observed.receipt?.state === 'pending' && observed.consumedGeneration === event.data.generation &&
    Number.isInteger(observed.consumedRequest) &&
    observed.consumedRequest < event.data.requestId;
  const deliver = (held, kind) => {
    if (!expectedActual(held.event, snapshot())) {
      state.problems.push(kind + ': held envelope no longer current'); record(kind + '-stale-refused', held.event); return;
    }
    record(kind + '-before', held.event);
    const actual = held.event;
    const copy = new MessageEvent('message', { data: actual.data, source: actual.source, origin: actual.origin });
    replay.add(copy); window.dispatchEvent(copy);
    record(kind + '-after', actual);
  };
  const messages = event => {
    if (replay.has(event) || !state.armed || event.data?.type !== 'noacg-editor-foundation-v1') return;
    const observed = snapshot(), part = event.data.parts?.find(part => part.selector === '#nested');
    if (!expectedActual(event, observed)) { record('ignored-not-current-envelope', event); return; }
    if (event.data.kind !== 'pose' || !part) { record('current-non-target-reply', event); return; }
    if (!sameRevision(part.appearance?.revision, event.data.revision) || part.appearance?.cue !== undefined) {
      state.problems.push('target part revision/cue differs from intended arrival pose'); record('wrong-part-pose', event); return;
    }
    if (!state.zero && !state.ruler && near(part.appearance?.time, 0) && near(observed.view.time, 0)) {
      if (!unconsumed(event, observed)) { state.problems.push('zero reply already consumed or not pending'); record('zero-ordering-failed', event); return; }
      event.stopImmediatePropagation();
      state.zero = { event, summary: clone(summary(event)), released: false };
      state.zeroHeldBeforeController = true; record('zero-stopped', event);
      queueMicrotask(() => {
        state.zeroHeldAfterMicrotask = !state.zero.released && unconsumed(event, snapshot()); record('zero-held-after-microtask', event);
      });
      later(() => {
        if (!state.zero.released) {
          state.problems.push('zero hold safety release before ruler'); state.zero.released = true;
          deliver(state.zero, 'zero-safety-release');
        }
      }, 4000);
    } else if (!state.one && state.ruler && state.zero?.released && near(part.appearance?.time, 1) && near(observed.view.time, 1)) {
      if (!unconsumed(event, observed) || event.data.documentId !== state.zero.summary.documentId ||
        event.data.generation !== state.zero.summary.generation || !sameRevision(event.data.revision, state.zero.summary.revision)) {
        state.problems.push('one reply ordering or zero/one lineage mismatch'); record('one-ordering-failed', event); return;
      }
      event.stopImmediatePropagation();
      state.one = { event, summary: clone(summary(event)), released: false };
      state.oneHeldBeforeController = true; record('one-stopped', event);
      queueMicrotask(() => { state.oneHeldAfterMicrotask = !state.one.released && unconsumed(event, snapshot()); record('one-held-after-microtask', event); });
      later(() => { state.one.released = true; state.latestReleased = true; deliver(state.one, 'one-release'); }, 1500);
    } else record('current-reply-not-held', event);
  };
  const clicks = event => {
    const button = event.target instanceof Element ? event.target.closest('button') : null;
    if (button?.getAttribute('aria-label') !== 'Enable Position X animation') return;
    if (!state.connected || !snapshot()) { state.problems.push('arming without connected diagnostics'); record('arm-unavailable'); return; }
    if (!state.armed) { state.armed = true; record('armed-by-aria-label'); }
  };
  const pointers = event => {
    if (!state.armed || !(event.target instanceof Element)) return;
    if (event.target.closest('[role="slider"][aria-label="Playhead"]')) {
      state.ruler = true; record('real-ruler-pointerdown');
      if (!state.zero || state.zero.released || !state.zeroHeldAfterMicrotask) { state.problems.push('ruler without validated held zero reply'); return; }
      later(() => {
        if (state.zero.released) return;
        state.zero.released = true; deliver(state.zero, 'zero-release-after-ruler');
        const observed = snapshot();
        state.zeroReplayConsumed = observed?.consumedRequest === state.zero.summary.requestId &&
          observed?.consumedGeneration === state.zero.summary.generation;
        record('zero-replay-consumption-checked');
      }, 0);
    }
    if (event.target.closest('[data-testid="foundation-canvas"]')) record('real-canvas-pointerdown');
  };
  // Serialized as a self-contained init script, before application modules exist.
  window.addEventListener('message', messages);
  document.addEventListener('click', clicks, true);
  document.addEventListener('pointerdown', pointers, true);
  state.stop = () => {
    window.removeEventListener('message', messages); document.removeEventListener('click', clicks, true);
    document.removeEventListener('pointerdown', pointers, true); timers.forEach(clearTimeout); timers.clear();
  };
  record('installed-before-app');
}
