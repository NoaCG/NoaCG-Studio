// Pure fake-EventTarget/DOM checks. No browser, server, queue or product import.
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import vm from 'node:vm';
const init = readFileSync('docs/work-specs/editor-keys-gate/prepared-probe-init.js.txt', 'utf8');
let checks = 0;
function check(name, fn) { fn(); checks++; process.stdout.write('PASS ' + name + '\n'); }
function harness({ late = false } = {}) {
  const listeners = new Map(), docListeners = new Map(), micro = [], timers = new Map();
  let now = 0, timerId = 0;
  const frame = {};
  const expected = () => ({ documentId: 'draft:stub', generation: 4, requestId: 10, revision: { source: 5, assets: 1 } });
  const current = { version: { source: 5, assets: 1 }, view: { time: 0, cue: null, selectedParts: ['#nested'] },
    receipt: { state: 'pending' }, expected: expected(), consumedRequest: 9, consumedGeneration: 4 };
  class Element {
    constructor(kind, label = '') { this.kind = kind; this.label = label; this.textContent = '◷'; }
    getAttribute(name) { return name === 'aria-label' ? this.label : null; }
    closest(selector) {
      if (selector === 'button') return this.kind === 'button' ? this : null;
      if (selector === '[role="slider"][aria-label="Playhead"]') return this.kind === 'ruler' ? this : null;
      if (selector === '[data-testid="foundation-canvas"]') return this.kind === 'canvas' ? this : null;
      return null;
    }
  }
  class MessageEvent {
    constructor(type, data) { Object.assign(this, data); this.type = type; this.stopped = false; }
    stopImmediatePropagation() { this.stopped = true; }
  }
  const add = (map, type, fn) => map.set(type, [...map.get(type) ?? [], fn]);
  const remove = (map, type, fn) => map.set(type, (map.get(type) ?? []).filter(f => f !== fn));
  const window = { top: null, addEventListener: (type, fn) => add(listeners, type, fn),
    removeEventListener: (type, fn) => remove(listeners, type, fn),
    dispatchEvent(event) { for (const fn of listeners.get(event.type) ?? []) { fn(event); if (event.stopped) break; } } };
  window.top = window;
  const document = { querySelector(selector) { return selector.startsWith('iframe') ? { contentWindow: frame } : { getAttribute: () => null }; },
    addEventListener: (type, fn) => add(docListeners, type, fn), removeEventListener: (type, fn) => remove(docListeners, type, fn) };
  const consume = event => {
    current.consumedRequest = event.data.requestId; current.consumedGeneration = event.data.generation;
    if (event.data.requestId === 10) current.expected = { ...expected(), requestId: 11 };
    if (event.data.requestId === 11) current.receipt = { state: 'ready', requestId: 11 };
  };
  if (late) window.addEventListener('message', consume);
  vm.runInNewContext('(' + init + ')()', { window, document, Element, MessageEvent,
    performance: { timeOrigin: 0, now: () => now }, queueMicrotask: fn => micro.push(fn),
    setTimeout(fn, ms) { const id = ++timerId; timers.set(id, { fn, at: now + ms }); return id; }, clearTimeout: id => timers.delete(id) });
  if (!late) window.addEventListener('message', consume);
  const state = window.__keysProbe;
  state.snapshot = () => current; state.connected = true;
  const flushMicro = () => { while (micro.length) micro.shift()(); };
  const advance = ms => {
    now += ms;
    for (const [id, timer] of [...timers]) if (timer.at <= now) { timers.delete(id); timer.fn(); flushMicro(); }
  };
  const event = (id = 10, time = 0, overrides = {}) => new MessageEvent('message', { source: frame, origin: 'null',
    data: { type: 'noacg-editor-foundation-v1', kind: 'pose', ...expected(), requestId: id,
      parts: [{ selector: '#nested', appearance: { time, revision: { source: 5, assets: 1 } } }], ...overrides } });
  const send = message => { window.dispatchEvent(message); flushMicro(); return message; };
  const doc = (type, target) => { for (const fn of docListeners.get(type) ?? []) fn({ type, target }); flushMicro(); };
  const arm = (label = 'Enable Position X animation') => doc('click', new Element('button', label));
  return { state, current, frame, event, send, doc, arm, advance, Element, window };
}
check('real button text ◷ arms by exact aria-label', () => { const h = harness(); h.arm(); assert.equal(h.state.armed, true); });
check('textContent equal to name does not arm with wrong aria-label', () => { const h = harness(); const button = new h.Element('button', 'other'); button.textContent = 'Enable Position X animation'; h.doc('click', button); assert.equal(h.state.armed, false); });
check('unarmed run has no successful held-zero preflight state', () => { const h = harness(); h.send(h.event()); assert.equal(h.state.zeroHeldBeforeController, false); });
check('unavailable diagnostics prevent arming', () => { const h = harness(); h.state.connected = false; h.arm(); assert.equal(h.state.armed, false); assert.equal(h.state.problems.length, 1); });
check('wrong iframe is not held', () => { const h = harness(); h.arm(); const e = h.event(); e.source = {}; h.send(e); assert.equal(h.state.zeroHeldBeforeController, false); });
for (const [name, override] of [['document', { documentId: 'other' }], ['generation', { generation: 7 }],
  ['revision', { revision: { source: 8, assets: 1 } }], ['request', { requestId: 99 }]]) {
  check('wrong ' + name + ' is not held', () => { const h = harness(); h.arm(); h.send(h.event(10, 0, override)); assert.equal(h.state.zeroHeldBeforeController, false); });
}
check('wrong selection is not held', () => { const h = harness(); h.arm(); h.current.view.selectedParts = ['#other']; h.send(h.event()); assert.equal(h.state.zeroHeldBeforeController, false); });
check('prior acknowledgement in wrong generation is not held', () => { const h = harness(); h.arm(); h.current.consumedGeneration = 3; h.send(h.event()); assert.equal(h.state.zeroHeldBeforeController, false); });
check('listener registered after consumer is rejected as ineffective', () => {
  const h = harness({ late: true }); h.arm(); h.send(h.event()); assert.equal(h.state.zeroHeldBeforeController, false); assert.ok(h.state.events.some(e => e.kind === 'ignored-not-current-envelope'));
});
check('early actual hold, microtask witness, original-payload replay and one hold', () => {
  const h = harness(); h.arm(); const zero = h.event(); h.send(zero);
  assert.equal(zero.stopped, true); assert.equal(h.current.consumedRequest, 9);
  assert.equal(h.state.zeroHeldBeforeController, true); assert.equal(h.state.zeroHeldAfterMicrotask, true);
  const recorded = h.state.events.find(e => e.kind === 'zero-stopped');
  h.current.view.time = 1; h.doc('pointerdown', new h.Element('ruler')); h.advance(0);
  assert.equal(h.current.consumedRequest, 10); assert.equal(h.state.zeroReplayConsumed, true);
  assert.equal(recorded.session.view.time, 0); // record is not a live mutable reference
  const one = h.event(11, 1); h.send(one);
  assert.equal(one.stopped, true); assert.equal(h.current.consumedRequest, 10);
  assert.equal(h.state.oneHeldBeforeController, true); assert.equal(h.state.oneHeldAfterMicrotask, true);
  h.doc('pointerdown', new h.Element('canvas'));
  const press = h.state.events.find(e => e.kind === 'real-canvas-pointerdown'); assert.equal(press.session.receipt.state, 'pending');
  h.advance(1500); assert.equal(h.current.consumedRequest, 11); assert.equal(h.state.latestReleased, true);
  assert.equal(h.state.events.find(e => e.kind === 'one-release-before').reply.requestId, one.data.requestId);
  h.state.stop();
});
check('replay refuses a held envelope that became stale', () => {
  const h = harness(); h.arm(); h.send(h.event()); h.current.view.time = 1; h.current.expected.requestId = 99;
  h.doc('pointerdown', new h.Element('ruler')); h.advance(0);
  assert.equal(h.current.consumedRequest, 9); assert.ok(h.state.problems.some(p => p.includes('held envelope no longer current')));
});
check('zero safety release reports ineffective and stays below controller watchdog', () => {
  const h = harness(); h.arm(); h.send(h.event()); h.advance(4000);
  assert.equal(h.state.zero.released, true); assert.ok(h.state.problems.includes('zero hold safety release before ruler'));
});
process.stdout.write('PASS ' + checks + ' pure stub checks; browser/runtime behavior not established\n');
