import { readFile, writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { decodePng } from '../scripts/png-decode.mjs';
const file=process.argv[2]||'f4-raw.json',r=JSON.parse(await readFile(new URL(file,import.meta.url),'utf8'));const summary={source:r.source,fixture:r.fixture,method:'Raw protocol input-to-ack timing is a proxy. Poses sample computed rendered bounds/opacity at rAF; distinct artwork crops from actual CDP PNG frames independently record visible changes. All frame gaps retained; no nominal frame-rate claim.',viewports:[]};
const stats=xs=>{const s=xs.filter(Number.isFinite).sort((a,b)=>a-b);return {n:s.length,median:s[Math.floor(s.length*.5)]??null,p95:s[Math.min(s.length-1,Math.ceil(s.length*.95)-1)]??null,max:s.at(-1)??null};};
const delta=xs=>xs.slice(1).map((x,i)=>x-xs[i]);
for(const v of r.viewports){const t={viewport:v.viewport,settings:v.settings,coldLoadMs:v.coldLoadMs,actions:[]};summary.viewports.push(t);for(const a of v.actions){const x={name:a.name,inputs:{pointerdown:a.events.filter(e=>e.kind==='pointerdown').length,pointerup:a.events.filter(e=>e.kind==='pointerup').length,pointermove:a.events.filter(e=>e.kind==='pointermove').length},protocolLatencyMs:stats(a.metrics.samples.map(s=>s.ms)),protocolByKind:Object.fromEntries([...new Set(a.metrics.samples.map(s=>s.kind))].map(k=>[k,stats(a.metrics.samples.filter(s=>s.kind===k).map(s=>s.ms))])),parentLongTasks:a.metrics.parentLongTasks,iframeLongTasks:a.metrics.longTasks,poseFrameIntervalMs:stats(delta(a.poses.map(p=>p.at))),screencastFrameIntervalMs:stats(delta(a.frames.map(p=>p.at))),intervals:[]};
 const changes=a.poses.filter((p,i)=>i===0||['x','y','w','h','opacity'].some(k=>p[k]!==a.poses[i-1][k]));x.changedPoseIntervalMs=stats(delta(changes.map(p=>p.at)));x.changedPoses=changes.length;
 const dir=new URL((file.includes('highrate')?'f4-highrate-':'f4-attempt2-')+v.viewport.width+'-'+a.name+'/',import.meta.url);let prev=null;const rendered=[];
 for(let i=0;i<a.frames.length;i++){const data=decodePng(await readFile(new URL(String(i).padStart(4,'0')+'.png',dir)));const b=a.artboard;const x0=Math.max(0,Math.round(b.x)),y0=Math.max(0,Math.round(b.y)),x1=Math.min(data.width,Math.round(b.x+b.width)),y1=Math.min(data.height,Math.round(b.y+b.height));const h=createHash('sha256');for(let y=y0;y<y1;y++)h.update(data.samples.subarray((y*data.width+x0)*data.channels,(y*data.width+x1)*data.channels));const hash=h.digest('hex');if(hash!==prev)rendered.push(a.frames[i].at);prev=hash;}
 x.renderedArtworkChanges=rendered.length;x.renderedArtworkChangeIntervalMs=stats(delta(rendered));
 const downs=a.events.filter(e=>e.kind==='pointerdown'),ups=a.events.filter(e=>e.kind==='pointerup');for(let i=0;i<downs.length;i++){const start=downs[i].at,end=ups.find(e=>e.at>=start)?.at??start;const moves=a.events.filter(e=>e.kind==='pointermove'&&e.at>=start&&e.at<=end);const pp=changes.filter(p=>p.at>=start&&p.at<=end),rr=rendered.filter(t=>t>=start&&t<=end);x.intervals.push({start,end,durationMs:end-start,moveEvents:moves.length,actualMoveHz:moves.length*1000/(end-start),changedPoses:pp.length,changedPoseHz:pp.length*1000/(end-start),renderedArtworkChanges:rr.length,renderedArtworkHz:rr.length*1000/(end-start),maxPoseGapMs:stats(delta([start,...pp.map(p=>p.at),end])).max,maxRenderedGapMs:stats(delta([start,...rr,end])).max,finalPointerUpProtocolProxyMs:a.metrics.samples.filter(s=>s.inputAt>=end&&s.inputAt<=end+1000).map(s=>s.ms)});}
 t.actions.push(x);await writeFile(new URL(file.includes('highrate')?'f4-highrate-summary.json':'f4-summary.json',import.meta.url),JSON.stringify(summary,null,2)+'\n');}}
console.log(JSON.stringify(summary.viewports.map(v=>({viewport:v.viewport,actions:v.actions.map(a=>({name:a.name,protocol:a.protocolLatencyMs,poses:a.changedPoses,painted:a.renderedArtworkChanges,motionHz:stats(a.intervals.filter(i=>i.durationMs>500).map(i=>i.actualMoveHz)),renderedHz:stats(a.intervals.filter(i=>i.durationMs>500).map(i=>i.renderedArtworkHz))}))})),null,2));
