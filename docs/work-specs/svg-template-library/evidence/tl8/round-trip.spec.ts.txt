/** Copied from e2e/svg-templates.spec.ts (at fac5edc): every visible text that overlaps another,
 *  is cut by a shape it is not inside, or is cut off by the frame edge; clipping windows (a credits
 *  roll, a crawl) count as clipped, not broken. */
async function airProblems(air: FrameLocator): Promise<string[]> {
  return air.locator('svg').first().evaluate((root) => {
    type Box = { l: number; t: number; r: number; b: number };
    const art = root.closest('svg') as SVGSVGElement;
    const frame = art.getBoundingClientRect();
    const shown = (el: Element): boolean => {
      for (let e: Element | null = el; e && e !== document.documentElement; e = e.parentElement) {
        const cs = getComputedStyle(e);
        if (cs.display === 'none' || cs.visibility === 'hidden' || Number(cs.opacity) === 0) return false;
      }
      return true;
    };
    const windowOf = (el: Element): Box | null => {
      for (let e = el.parentElement; e && e !== art; e = e.parentElement) {
        const clipRef = /url\(#(noacg-crawl-[^)]+)\)/.exec(e.getAttribute('clip-path') ?? '')?.[1];
        const clipShape = clipRef ? art.querySelector(`[id="${clipRef}"] polygon`) as SVGPolygonElement | null : null;
        if (clipShape && e instanceof SVGGraphicsElement) {
          const m = e.getScreenCTM();
          if (!m || !clipShape.points.numberOfItems) return null;
          const p = Array.from({ length: clipShape.points.numberOfItems }, (_, i) => new DOMPoint(clipShape.points.getItem(i).x, clipShape.points.getItem(i).y).matrixTransform(m));
          return { l: Math.min(...p.map((q) => q.x)), t: Math.min(...p.map((q) => q.y)), r: Math.max(...p.map((q) => q.x)), b: Math.max(...p.map((q) => q.y)) };
        }
        if (e instanceof SVGSVGElement && e.parentElement instanceof SVGGraphicsElement) {
          const m = (e.parentElement as SVGGraphicsElement).getScreenCTM();
          if (!m) return null;
          const x = e.x.baseVal.value, y = e.y.baseVal.value, w = e.width.baseVal.value, h = e.height.baseVal.value;
          const p = [new DOMPoint(x, y).matrixTransform(m), new DOMPoint(x + w, y + h).matrixTransform(m)];
          return { l: Math.min(p[0].x, p[1].x), t: Math.min(p[0].y, p[1].y), r: Math.max(p[0].x, p[1].x), b: Math.max(p[0].y, p[1].y) };
        }
      }
      return null;
    };
    const cut = (r: Box, w: Box): Box | null => {
      const c = { l: Math.max(r.l, w.l), t: Math.max(r.t, w.t), r: Math.min(r.r, w.r), b: Math.min(r.b, w.b) };
      return c.r - c.l > 1 && c.b - c.t > 1 ? c : null;
    };
    const clip = (el: Element, r: Box): Box | null => {
      const w = windowOf(el);
      return w ? cut(r, w) : r;
    };
    const boxOf = (el: Element, inset = 0): Box | null => {
      const g = el.getBoundingClientRect();
      if (!(g.width > 0 && g.height > 0) || !shown(el)) return null;
      const c = g.height * inset;
      return clip(el, { l: g.left, t: g.top + c, r: g.right, b: g.bottom - c });
    };
    const cross = (a: Box, b: Box) => Math.min(a.r, b.r) - Math.max(a.l, b.l) > 2 && Math.min(a.b, b.b) - Math.max(a.t, b.t) > 2;
    const holds = (o: Box, i: Box) => o.l <= i.l + 2 && o.t <= i.t + 2 && o.r >= i.r - 2 && o.b >= i.b - 2;
    const name = (el: Element) => `"${(el.textContent ?? '').trim().slice(0, 30)}"`;
    const texts = [...art.querySelectorAll('text')]
      .filter((t) => (t.textContent ?? '').trim() && !t.closest('[class$="-credits"]'))
      .map((el) => ({ el, box: boxOf(el, 1 / 6), rolled: !!windowOf(el) }))
      .filter((t): t is { el: SVGTextElement; box: Box; rolled: boolean } => !!t.box);
    const shapes = [...art.querySelectorAll('rect, path, polygon, polyline, line, ellipse, circle')]
      .map((el) => ({ el, box: boxOf(el) }))
      .filter((s): s is { el: Element; box: Box } => !!s.box);
    const out: string[] = [];
    texts.forEach((a, i) => {
      const fr = { l: frame.left, t: frame.top, r: frame.right, b: frame.bottom };
      const seen = cut(a.box, fr);
      if (seen && !a.rolled && !holds(fr, a.box)) out.push(`${name(a.el)} is cut off by the frame edge`);
      for (const b of texts.slice(i + 1)) if (cross(a.box, b.box)) out.push(`${name(a.el)} overlaps ${name(b.el)}`);
      const covered = (s: Element) => shapes.some((h) => h.el !== s && holds(h.box, a.box) && !!(s.compareDocumentPosition(h.el) & Node.DOCUMENT_POSITION_FOLLOWING));
      for (const s of shapes) if (cross(a.box, s.box) && !holds(s.box, a.box) && !covered(s.el)) out.push(`${name(a.el)} is cut by a <${s.el.tagName}>${s.el.id ? ' #' + s.el.id : ''}`);
    });
    return out;
  });
}

async function onAir(page: Page, air: FrameLocator, label: string, settle = 1_200): Promise<void> {
  // Recorded, not thrown: a finding is reported and the walk goes on to the remaining steps.
  let found: string[] = [];
  await expect.poll(async () => (found = await airProblems(air)), { timeout: 6_000 }).toEqual([]).catch(() => undefined);
  problems[label] = found;
  if (settle) await page.waitForTimeout(settle); // the entrance finishes before the picture
  await page.getByTestId('program-stage').screenshot({ path: `${SHOTS}/${label}.png` });
}

/** What the restyle put on air: the font a text is set in (and whether that face is loaded in the
 *  output frame), and the first plate's fill. */
async function looks(air: FrameLocator, text: string) {
  return air.locator('svg').first().evaluate(async (root, t) => {
    const el = [...root.querySelectorAll('text')].find((e) => (e.textContent ?? '').includes(t));
    const plate = root.querySelector('rect');
    await document.fonts.ready;
    const font = el ? getComputedStyle(el).fontFamily : null;
    return { text: t, font, outfitLoaded: document.fonts.check("800 40px 'Outfit'") || document.fonts.check('800 40px Outfit'), plate: plate ? getComputedStyle(plate).fill : null };
  }, text);
}

async function fieldFor(air: FrameLocator, title: string): Promise<string> {
  const field = await air.locator('svg').first().evaluate((_, n) => {
    const d = (window as unknown as { SPXGCTemplateDefinition: { DataFields: { field: string; title: string }[] } }).SPXGCTemplateDefinition;
    return d.DataFields.find((e) => e.title === n)?.field;
  }, title);
  expect(field, `the operator has a ${title} field`).toBeTruthy();
  return field!;
}

/** Every graphic of the production open on the page, with its cues' values: one cue each, never
 *  a blank or a second one. */
async function cuesOf(page: Page) {
  return page.evaluate(async () => {
    const { loadShows } = await __noacgImport('/src/model/shows.ts');
    const id = new URL(location.href).searchParams.get('show') ?? location.pathname.split('/').pop();
    const shows = loadShows();
    const show = shows.find((s) => s.id === id) ?? shows[shows.length - 1];
    return { production: show.name, graphics: show.graphics.map((g) => ({ name: g.name, cues: (show.cues ?? []).filter((c) => c.sourceId === g.id).length })) };
  });
}

async function importInto(page: Page, slug: string, graphic: string, production: string, kind: string, opts: { existing?: boolean; clock?: boolean; credits?: boolean } = {}) {
  await page.goto('/app');
  await dropSvg(page, restyled(slug));
  await expect(page.getByTestId('map-svg-fields')).toBeVisible();
  const proposed: Record<string, unknown> = {};
  if (opts.credits) {
    proposed.creditsEngine = await page.getByTestId('map-svg-credits-engine').inputValue();
    proposed.creditsLayout = await page.getByTestId('map-svg-credits-preview').getAttribute('data-layout');
    proposed.creditsList = (await page.getByTestId('map-svg-credits-list').textContent())?.slice(0, 120);
    expect(proposed.creditsEngine).toBe('editable');
  } else {
    proposed.kind = await page.getByTestId('map-svg-behaviour-kind').inputValue();
    proposed.missing = await page.getByTestId('map-svg-behaviour-missing').count();
    expect(proposed.kind, `${slug}: proposed behaviour`).toBe(kind);
    expect(proposed.missing, `${slug}: no missing-layer warning`).toBe(0);
  }
  if (opts.clock) {
    proposed.clockModeOffered = await page.getByTestId('map-svg-clock-mode').inputValue();
    await page.getByTestId('map-svg-clock-mode').selectOption('duration');
  }
  // The fonts the import found, as the Fonts section reads them.
  proposed.fonts = await page.locator('body').evaluate(() => [...document.querySelectorAll('[data-testid^="map-svg-font"]')].map((e) => (e.textContent ?? '').replace(/\s+/g, ' ').trim()).filter(Boolean).slice(0, 6));
  if (opts.existing) await intoExistingProduction(page, graphic, production);
  else await intoProduction(page, graphic, production);
  await settleDurableWrites(page);
  return proposed;
}

test('1. download the nine TL8 templates as a user does', async ({ page }) => {
  test.setTimeout(300_000);
  mkdirSync(path.join(DIR, 'downloaded'), { recursive: true });
  await page.goto('/svg-templates');
  const report: string[] = [];
  for (const t of ALL) {
    const tile = page.getByTestId(`svg-package-${t.pkg}-${t.slug}`);
    await tile.scrollIntoViewIfNeeded();
    const [dl] = await Promise.all([page.waitForEvent('download'), tile.locator('a[download]').first().click()]);
    const to = path.join(DIR, 'downloaded', `${t.slug}.svg`);
    await dl.saveAs(to);
    expect(readFileSync(to).equals(source(t.slug)), `${t.name}: the download is the source file`).toBe(true);
    report.push(`${t.slug}.svg per-file download from the ${t.pkg} package tile (${dl.url().replace(/^https?:\/\/[^/]+/, '')}) = source`);
  }
  for (const pkg of [...new Set(ALL.map((t) => t.pkg))]) {
    // A fresh page per zip: after ten downloads from one page Chromium holds further ones for a
    // multiple-downloads permission (a browser rule, not the site).
    await page.goto('/svg-templates');
    const [zip] = await Promise.all([page.waitForEvent('download'), page.getByTestId(`svg-package-${pkg}-zip`).click()]);
    const zipTo = path.join(DIR, `package-${pkg}.zip`);
    await zip.saveAs(zipTo);
    const unz = path.join(DIR, 'zip', pkg);
    mkdirSync(unz, { recursive: true });
    execFileSync('unzip', ['-o', '-q', zipTo, '-d', unz]);
    const files = execFileSync('find', [unz, '-name', '*.svg']).toString().trim().split('\n');
    for (const t of ALL.filter((x) => x.pkg === pkg)) {
      const hit = files.find((f) => path.basename(f) === `${t.slug}.svg`);
      expect(hit, `${t.slug}.svg is in the ${pkg} zip`).toBeTruthy();
      expect(readFileSync(hit!).equals(source(t.slug)), `${t.name}: the ${pkg} zip copy is the source file`).toBe(true);
      report.push(`${path.relative(path.join(DIR, 'zip'), hit!)} in ${zip.url().replace(/^https?:\/\/[^/]+/, '')} = source`);
    }
  }
  writeFileSync(path.join(DIR, 'download-report.txt'), report.join('\n') + '\n');
});

test('2b. games and clocks: Countdown and Show clock', async ({ page }) => {

    const w = window as unknown as { gsap: { getProperty(el: Element, p: string): number } };

    return { move: move ? move.name : null, active: move ? move.tween.isActive() : false, opacity: Number(w.gsap.getProperty(el, 'opacity')), scale: Number(w.gsap.getProperty(el, 'scale')) };
  });

  const boxes = page.locator('input[data-testid^="cue-field-"]');
  let minutes = '';
  for (let i = 0; i < await boxes.count(); i++) if ((await boxes.nth(i).inputValue()) === '1') minutes = (await boxes.nth(i).getAttribute('data-testid'))!;
  log.countdownMinutesField = minutes;

  await page.getByTestId(minutes).fill('0.25'); // fifteen seconds

  log.countdownPulse = { minOpacityOverOneSecond: Math.min(...dims), clock: await air.locator('.imported-design-clock').textContent() };

  await onAir(page, air, 'countdown-3-last-ten-seconds', 0);
  await expect(role('expired')).toHaveClass(/imported-design-on/, { timeout: 15_000 });
